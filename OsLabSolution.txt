1.FCFS

#include <stdio.h>

int main() {
    int n, at[10], bt[10], ct[10], tat[10], wt[10];

    printf("Enter number of processes: ");
    scanf("%d", &n);

    printf("Enter Arrival Time and Burst Time:\n");
    for(int i = 0; i < n; i++) {
        scanf("%d %d", &at[i], &bt[i]);
    }

    // 🔹 Sort by Arrival Time (FCFS order)
    for(int i = 0; i < n - 1; i++) {
        for(int j = i + 1; j < n; j++) {
            if(at[i] > at[j]) {
                // swap AT
                int temp = at[i];
                at[i] = at[j];
                at[j] = temp;

                // swap BT
                temp = bt[i];
                bt[i] = bt[j];
                bt[j] = temp;
            }
        }
    }

    // 🔹 Completion Time
    ct[0] = at[0] + bt[0];

    for(int i = 1; i < n; i++) {
        if(ct[i-1] < at[i]) {
            ct[i] = at[i] + bt[i];   // CPU idle
        } else {
            ct[i] = ct[i-1] + bt[i];
        }
    }

    // 🔹 TAT and WT
    for(int i = 0; i < n; i++) {
        tat[i] = ct[i] - at[i];
        wt[i] = tat[i] - bt[i];
    }

    // 🔹 Output
    printf("\nP\tAT\tBT\tCT\tTAT\tWT\n");
    for(int i = 0; i < n; i++) {
        printf("%d\t%d\t%d\t%d\t%d\t%d\n",
               i+1, at[i], bt[i], ct[i], tat[i], wt[i]);
    }

    return 0;
}

1.SJF(PREEMPTIVE)

#include <stdio.h>

int main() {
    int n;
    int at[10], bt[10], rt[10];
    int wt[10], tat[10];
    int complete = 0, time = 0, min, shortest = -1;
    int finish;

    printf("Enter number of processes: ");
    scanf("%d", &n);

    printf("Enter Arrival Time and Burst Time:\n");
    for(int i = 0; i < n; i++) {
        scanf("%d %d", &at[i], &bt[i]);
        rt[i] = bt[i];  // remaining time
    }

    while(complete != n) {
        min = 9999;
        shortest = -1;

        for(int i = 0; i < n; i++) {
            if(at[i] <= time && rt[i] > 0 && rt[i] < min) {
                min = rt[i];
                shortest = i;
            }
        }

        if(shortest == -1) {
            time++;
            continue;
        }

        rt[shortest]--;
        time++;

        if(rt[shortest] == 0) {
            complete++;
            finish = time;

            wt[shortest] = finish - bt[shortest] - at[shortest];
            if(wt[shortest] < 0)
                wt[shortest] = 0;

            tat[shortest] = wt[shortest] + bt[shortest];
        }
    }

    printf("\nP\tAT\tBT\tWT\tTAT\n");
    for(int i = 0; i < n; i++) {
        printf("%d\t%d\t%d\t%d\t%d\n", i+1, at[i], bt[i], wt[i], tat[i]);
    }

    return 0;
}

1.Round Robin

#include <stdio.h>

int main() {
    int n, tq;
    int at[10], bt[10], rem[10], wt[10], tat[10], ct[10];
    int time = 0, completed = 0;

    printf("Enter number of processes: ");
    scanf("%d", &n);

    printf("Enter Arrival Time and Burst Time:\n");
    for(int i = 0; i < n; i++) {
        scanf("%d %d", &at[i], &bt[i]);
        rem[i] = bt[i];
    }

    printf("Enter Time Quantum: ");
    scanf("%d", &tq);

    while(completed < n) {
        int executed = 0;

        for(int i = 0; i < n; i++) {
            if(rem[i] > 0 && at[i] <= time) {

                executed = 1;

                if(rem[i] > tq) {
                    time += tq;
                    rem[i] -= tq;
                } else {
                    time += rem[i];
                    ct[i] = time;
                    rem[i] = 0;
                    completed++;

                    tat[i] = ct[i] - at[i];
                    wt[i] = tat[i] - bt[i];
                }
            }
        }

        // If no process was executed → CPU idle
        if(!executed) {
            time++;
        }
    }

    printf("\nP\tAT\tBT\tCT\tWT\tTAT\n");
    for(int i = 0; i < n; i++) {
        printf("%d\t%d\t%d\t%d\t%d\t%d\n",
               i+1, at[i], bt[i], ct[i], wt[i], tat[i]);
    }

    return 0;
}

            tat[highest] = wt[highest] + bt[highest];
        }
    }

    printf("\nP\tAT\tBT\tPR\tWT\tTAT\n");

3.DeadLock Prevention

#include <stdio.h>

int main() {
    int n, m;

    printf("Enter number of processes: ");
    scanf("%d", &n);

    printf("Enter number of resource types: ");
    scanf("%d", &m);

    int alloc[n][m], max[n][m], need[n][m], avail[m];

    printf("\nEnter Allocation Matrix:\n");
    for(int i = 0; i < n; i++) {
        for(int j = 0; j < m; j++) {
            scanf("%d", &alloc[i][j]);
        }
    }

    printf("\nEnter Max Matrix:\n");
    for(int i = 0; i < n; i++) {
        for(int j = 0; j < m; j++) {
            scanf("%d", &max[i][j]);
        }
    }

    printf("\nEnter Available Resources:\n");
    for(int i = 0; i < m; i++) {
        scanf("%d", &avail[i]);
    }

    // Calculate Need matrix
    for(int i = 0; i < n; i++) {
        for(int j = 0; j < m; j++) {
            need[i][j] = max[i][j] - alloc[i][j];
        }
    }

    int finish[n], safeSeq[n];
    for(int i = 0; i < n; i++)
        finish[i] = 0;

    int count = 0;

    while(count < n) {
        int found = 0;

        for(int i = 0; i < n; i++) {
            if(finish[i] == 0) {
                int j;
                for(j = 0; j < m; j++) {
                    if(need[i][j] > avail[j])
                        break;
                }

                if(j == m) {
                    for(int k = 0; k < m; k++) {
                        avail[k] += alloc[i][k];
                    }

                    safeSeq[count++] = i;
                    finish[i] = 1;
                    found = 1;
                }
            }
        }

        if(found == 0) {
            printf("\nSystem is NOT in safe state!\n");
            return 0;
        }
    }

    printf("\nSystem is in SAFE state.\nSafe sequence: ");
    for(int i = 0; i < n; i++) {
        printf("P%d ", safeSeq[i]);
    }

    return 0;
}

4.Producer Consumer

#include <stdio.h>
#include <stdlib.h>

int m = 1, f = 0, e = 3, i = 0;

int wait(int s) {
    return (s - 1);
}

int signal(int s) {
    return (s + 1);
}

void producer() {
    m = wait(m);
    f = signal(f);
    e = wait(e);

    i = i + 1;
    printf("Producer produced item %d\n", i);

    m = signal(m);
}

void consumer() {
    m = wait(m);
    f = wait(f);
    e = signal(e);

    printf("Consumer consumed item %d\n", i);
    i = i - 1;

    m = signal(m);
}

int main() {
    int choice;

    printf("1. Producer\n2. Consumer\n3. Exit\n");

    while (1) {
        printf("Enter choice: ");
        scanf("%d", &choice);

        switch (choice) {
            case 1:
                if (e != 0) {
                    producer();
                } else {
                    printf("Buffer is full\n");
                }
                break;

            case 2:
                if (f != 0) {
                    consumer();
                } else {
                    printf("Buffer is empty\n");
                }
                break;

            case 3:
                exit(0);

            default:
                printf("Invalid choice\n");
        }
    }

    return 0;
}

struct msgbuf {
    long type;
    char text[50];
};

int main() {
    int msgid = msgget(IPC_PRIVATE, 0666 | IPC_CREAT);

    struct msgbuf msg;

    if (fork() == 0) {
        msg.type = 1;
        sprintf(msg.text, "Hello Message Queue");
        msgsnd(msgid, &msg, sizeof(msg.text), 0);
    } else {
        msgrcv(msgid, &msg, sizeof(msg.text), 1, 0);
        printf("Received: %s\n", msg.text);
    }

    return 0;

#include <stdio.h>

int main() {
    int frames, pages, faults = 0;
    int ref[20], frame[10];

    printf("Enter number of frames: ");
    scanf("%d", &frames);

    printf("Enter number of pages: ");
    scanf("%d", &pages);

    printf("Enter reference string:\n");
    for(int i = 0; i < pages; i++) {
        scanf("%d", &ref[i]);
    }

    for(int i = 0; i < frames; i++) {
        frame[i] = -1;
    }

    int index = 0;

    for(int i = 0; i < pages; i++) {
        int found = 0;

        for(int j = 0; j < frames; j++) {
            if(frame[j] == ref[i]) {
                found = 1;
                break;
            }
        }

        if(!found) {
            frame[index] = ref[i];
            index = (index + 1) % frames;
            faults++;
        }

        printf("\nFrames: ");
        for(int j = 0; j < frames; j++) {
            printf("%d ", frame[j]);
        }
    }

    printf("\nPage Faults = %d\n", faults);

    return 0;
}

    int frames, pages, faults = 0;
    int ref[20], frame[10], time[10];

    printf("Enter number of frames: ");
    scanf("%d", &frames);

    printf("Enter number of pages: ");

        scanf("%d", &ref[i]);
    }

    for(int i = 0; i < frames; i++) {
        frame[i] = -1;
        time[i] = 0;
    }

    int counter = 0;

    for(int i = 0; i < pages; i++) {
        int found = 0;

        for(int j = 0; j < frames; j++) {
            if(frame[j] == ref[i]) {
                counter++;
                time[j] = counter;
                found = 1;
                break;
            }
        }

        if(!found) {
            int min = time[0], pos = 0;

            for(int j = 1; j < frames; j++) {
                if(time[j] < min) {
                    min = time[j];
                    pos = j;
                }
            }

            frame[pos] = ref[i];
            counter++;
            time[pos] = counter;
            faults++;
        }

        printf("\nFrames: ");
        for(int j = 0; j < frames; j++) {
            printf("%d ", frame[j]);
        }
    }

    printf("\nPage Faults = %d\n", faults);

    return 0;
}

                break;
            }
        }

        if(!found) {
            int farthest = -1, pos = -1;

            for(int j = 0; j < frames; j++) {
                int k;
                for(k = i + 1; k < pages; k++) {
                    if(frame[j] == ref[k]) {
                        break;
                    }
                }

                if(k > farthest) {
                    farthest = k;
                    pos = j;
                }
            }

            frame[pos] = ref[i];
            faults++;
        }

        printf("\nFrames: ");
        for(int j = 0; j < frames; j++) {
            printf("%d ", frame[j]);
        }
    }

    printf("\nPage Faults = %d\n", faults);

    return 0;
}
